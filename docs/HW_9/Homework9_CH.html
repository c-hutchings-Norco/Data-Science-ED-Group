mysummary <- function(data, variable){
  summary <- data |>
    group_by(get(variable)) |>
    summarize(n = n(),
              across(where(is.numeric),
                     list("mean" = mean, "median" = median, "sd" = sd, "IQR" = IQR),
                     .names = "{.fn}_{.col}"))
  colnames(summary)[1] <- variable
  return(summary)
}

Standardize the variables by calculating Z-scores

set.seed(10)
bike_split <- rsample::initial_split(bike_daily, prop = 0.75, strata = Seasons)
training <- training(bike_split)
testing <- testing(bike_split)


# Standardized numeric variables for the test and training sets

std_training <- training |>
  ungroup() |>
  mutate(
    across(
      where(is.numeric) & !all_of("RentedBikeCount"),
      ~ (.x - mean(.x, na.rm = TRUE)) /
        sd(.x, na.rm = TRUE)
    )
  )

train_means <- training |>
  ungroup() |> 
  summarise(
    across(
      where(is.numeric) & !all_of("RentedBikeCount"),
      ~ mean(.x, na.rm = TRUE)
    )
  )

train_sds <- training |>
  ungroup() |> 
  summarise(
    across(
      where(is.numeric) & !all_of("RentedBikeCount"),
      ~ sd(.x, na.rm = TRUE)
    )
  )

std_testing <- testing |>
  ungroup() |>
  mutate(
    across(
      where(is.numeric) & !all_of("RentedBikeCount"),
      ~ (.x - train_means[[cur_column()]]) /
        train_sds[[cur_column()]]
    )
  )
