Create summary statistics of numerical variables across categorical variables.

mysummary <- function(data, variable){
  summary <- data |>
    group_by(get(variable)) |>
    summarize(n = n(),
              across(where(is.numeric),
                     list("mean" = mean, "median" = median, "sd" = sd, "IQR" = IQR),
                     .names = "{.fn}_{.col}"))
  colnames(summary)[1] <- variable
  return(summary)
}

# Create a function to output weekday or weekend as a factor variable
date_factor <- function(Date) {
  wend_wday <- wday(Date,
       label = TRUE)
  weekend <- ifelse(wend_wday %in% c("Sat", "Sun"), "Weekend", "Weekday")
  return(as.factor(weekend))
}

# Modify training data set
training <- training |>
  mutate(Weekend = date_factor(Date))
