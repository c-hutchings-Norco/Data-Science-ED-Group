  1. Create a function to queary API. Function will include title/ subject to search, time period to search from, and API key Function Creation:
search_news <- function(title, time, key){
  result <- httr::GET(paste0("https://newsapi.org/v2/everything?q=",
                             title,
                             "&from=",
                             time,
                             "&language=en&pageSize=50&apiKey=",
                             key
))
  parsed <- jsonlite::fromJSON(rawToChar(result$content))
  return(dplyr::as_tibble(parsed$articles))
}
